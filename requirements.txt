# eyv_poller + jarvis_core — dependencies
#
#   pip install -r requirements.txt

requests
anthropic
PyQt6            # jarvis_core/dashboard.py and jarvis_core/voice/ui.py only —
                 # not needed to run eyv_poller or jarvis_core's own reasoning/polling process
faster-whisper   # jarvis_core/voice/ — local speech-to-text (no cloud STT dependency)
sounddevice      # jarvis_core/voice/ — microphone capture + audio playback
psutil           # jarvis_core/voice/ui.py — CPU/RAM stats in the HUD's left panel
face-recognition # jarvis_core/voice/face_id.py — face enrollment/identification (wraps dlib)
opencv-python    # jarvis_core/voice/face_id.py — webcam capture for face_recognition
vosk             # jarvis_core/voice/stt_vosk.py — alternate offline STT engine (JARVIS_VOICE_STT_ENGINE=vosk)
edge-tts         # jarvis_core/voice/tts_edge.py — alternate free/online TTS engine (JARVIS_VOICE_TTS_ENGINE=edgetts)
pydub            # jarvis_core/voice/tts_edge.py — decodes EdgeTTS's MP3 response to raw PCM; needs a real
                 # `ffmpeg` binary installed and on PATH (not a pip package) to actually work
audioop-lts; python_version >= "3.13"  # pydub imports the stdlib `audioop` module, which PEP 594
                 # removed in Python 3.13 — this backport restores it. Without it, `import pydub`
                 # itself fails on 3.13+ (confirmed: present as a Windows-Python-3.14 built-in on
                 # this dev machine, but genuinely absent on the Linux 3.14 GitHub Actions runner).
